% ==================== BƯỚC 1: CHUẨN BỊ MÔI TRƯỜNG ====================
\section{BƯỚC 1: Chuẩn bị môi trường phần mềm trên Windows}

Để khởi chạy dự án trực tiếp trên Windows, máy tính của bạn cần được cài đặt sẵn các phần mềm sau:

\begin{stepbox}{Danh sách các công cụ cần có trên máy}
\begin{itemize}[leftmargin=5mm,itemsep=2pt]
  \item \textbf{PHP 8.2+ (hoặc PHP 8.3):} Bật các extension cần thiết trong file \texttt{php.ini} (\codeinline{pdo\_mysql}, \codeinline{mbstring}, \codeinline{openssl}, \codeinline{curl}, \codeinline{fileinfo}, \codeinline{mongodb}).
  \item \textbf{Composer:} Trình quản lý gói cho PHP (Tải tại: \url{https://getcomposer.org/}).
  \item \textbf{Node.js (LTS v20 hoặc v22) \& npm:} Trình thực thi JavaScript (Tải tại: \url{https://nodejs.org/}).
  \item \textbf{Cơ sở dữ liệu MySQL 8.0:} Bạn có thể dùng qua phần mềm \textbf{XAMPP}, \textbf{Laragon}, \textbf{WampServer} hoặc cài trực tiếp MySQL Community Server.
  \item \textbf{Cơ sở dữ liệu MongoDB 8.0 Community Server:} Cài đặt và bật dịch vụ MongoDB trên cổng mặc định \texttt{27017}.
  \item \textbf{Redis Server (Tùy chọn/Khuyên dùng):} Dành cho tính năng hàng đợi và broadcast thời gian thực.
\end{itemize}
\end{stepbox}

% ==================== BƯỚC 2: TẢI MÃ NGUỒN ====================
\section{BƯỚC 2: Tải mã nguồn dự án từ GitHub}

Mở cửa sổ dòng lệnh (\textbf{Command Prompt} hoặc \textbf{PowerShell}) tại thư mục bạn muốn lưu dự án (ví dụ: \texttt{D:\textbackslash} hoặc Desktop) và chạy các lệnh sau:

\begin{terminalcmd}{Tải mã nguồn từ GitHub và chuyển vào thư mục dự án}
git clone https://github.com/hongphuoc6104/quanlykhachsan.git
cd quanlykhachsan
\end{terminalcmd}

% ==================== BƯỚC 3: THIẾT LẬP DỰ ÁN TỰ ĐỘNG ====================
\section{BƯỚC 3: Khởi tạo và thiết lập dự án tự động (Chạy file setup.bat)}

Để tiết kiệm thời gian và tránh các thao tác dòng lệnh phức tạp, dự án đã chuẩn bị sẵn file kịch bản tự động hóa \textbf{\texttt{setup.bat}}.

\begin{stepbox}{Thao tác thiết lập toàn bộ hệ thống trong 1 bước}
\begin{enumerate}[leftmargin=5mm,itemsep=2pt]
  \item Đảm bảo dịch vụ \textbf{MySQL} (trên XAMPP/Laragon) và \textbf{MongoDB} đã được bật trên máy tính.
  \item Mở thư mục dự án \texttt{quanlykhachsan} và \textbf{nhấp đúp chuột (Double-click)} vào file \textbf{\texttt{setup.bat}} (hoặc gõ \codeinline{setup.bat} trong cửa sổ dòng lệnh).
  \item File script sẽ tự động thực hiện hoàn toàn các công việc sau từ A đến Z:
    \begin{itemize}[itemsep=1.5pt]
      \item Kiểm tra môi trường PHP, Composer, Node.js và npm trên máy.
      \item Tự động khởi tạo file cấu hình \texttt{.env} cho cả Backend và Frontend từ file mẫu.
      \item Cài đặt toàn bộ các gói thư viện PHP qua Composer (\texttt{composer install}).
      \item Sinh khóa bảo mật ứng dụng (\codeinline{php artisan key:generate}).
      \item Tạo liên kết thư mục lưu trữ hình ảnh (\codeinline{php artisan storage:link}).
      \item Tự động chạy lệnh tạo bảng CSDL (\textit{Migrations}) và nạp sẵn toàn bộ dữ liệu mẫu (\textit{Seeders}).
      \item Cài đặt toàn bộ dependencies Node.js cho Realtime Service và Frontend Vue 3.
    \end{itemize}
\end{enumerate}
\end{stepbox}

% ==================== BƯỚC 4: KHỞI CHẠY DỰ ÁN ====================
\section{BƯỚC 4: Khởi chạy toàn bộ hệ thống}

Sau khi chạy \texttt{setup.bat} hoàn tất, bạn có thể khởi chạy hệ thống bằng 1 trong 2 cách sau:

\subsection*{Cách 1: Nhấp đúp vào file \texttt{start.bat} (Khuyên dùng -- Tự động mở cả 3 dịch vụ)}
\begin{itemize}[leftmargin=5mm,itemsep=1.5pt]
  \item \textbf{Nhấp đúp chuột trái} vào file \textbf{\texttt{start.bat}} trong thư mục dự án.
  \item Script sẽ tự động mở 3 cửa sổ dòng lệnh riêng biệt để chạy song song:
    \begin{enumerate}[itemsep=1pt]
      \item \textbf{Cửa sổ 1:} Backend Laravel 12 API (Cổng 8000).
      \item \textbf{Cửa sổ 2:} Realtime WebSocket Server (Cổng 3001).
      \item \textbf{Cửa sổ 3:} Frontend Vue 3 Development Server (Cổng 3000).
    \end{enumerate}
\end{itemize}

\subsection*{Cách 2: Khởi chạy thủ công từ 3 cửa sổ dòng lệnh riêng biệt}
Nếu bạn muốn tự tay quản lý từng tiến trình, hãy mở 3 cửa sổ CMD tại thư mục \texttt{quanlykhachsan} và chạy:

\begin{terminalcmd}{Cửa sổ 1: Chạy máy chủ Backend API (Laravel 12)}
cd backend
php artisan serve
\end{terminalcmd}

\begin{terminalcmd}{Cửa sổ 2: Chạy máy chủ thời gian thực (Socket.IO)}
cd realtime
npm start
\end{terminalcmd}

\begin{terminalcmd}{Cửa sổ 3: Chạy giao diện Web Frontend (Vue 3)}
cd frontend
npm run dev
\end{terminalcmd}

% ==================== BƯỚC 6: XỬ LÝ LỖI THƯỜNG GẶP ====================
\section{BƯỚC 6: Hướng dẫn xử lý các sự cố thường gặp trên Windows}

\begin{warningbox}[1. Lỗi: "could not find driver" khi chạy php artisan migrate]
\textbf{Nguyên nhân:} PHP chưa bật extension kết nối cơ sở dữ liệu MySQL.\\
\textbf{Cách xử lý:} Mở file \texttt{php.ini} (trong thư mục cài đặt PHP hoặc XAMPP), tìm dòng \texttt{;extension=pdo\_mysql} và xóa dấu chấm phẩy \texttt{;} ở đầu dòng thành \codeinline{extension=pdo\_mysql}, sau đó lưu file lại.
\end{warningbox}

\begin{warningbox}[2. Lỗi: "Class 'MongoDB\textbackslash Driver\textbackslash Manager' not found"]
\textbf{Nguyên nhân:} PHP chưa cài đặt extension MongoDB trên Windows.\\
\textbf{Cách xử lý:} Tải file \texttt{php\_mongodb.dll} tương ứng với phiên bản PHP từ PECL (\url{https://pecl.php.net/package/mongodb}), chép vào thư mục \texttt{php/ext/}, sau đó thêm dòng \codeinline{extension=mongodb} vào file \texttt{php.ini}.
\end{warningbox}

\begin{warningbox}[3. Lỗi: "Access denied for user 'root'@'localhost'"]
\textbf{Nguyên nhân:} Sai thông tin tài khoản hoặc mật khẩu kết nối MySQL.\\
\textbf{Cách xử lý:} Mở file \texttt{backend/.env}, kiểm tra lại các thông số \codeinline{DB\_USERNAME} và \codeinline{DB\_PASSWORD} cho trùng khớp với tài khoản MySQL trên máy tính của bạn.
\end{warningbox}

\begin{warningbox}[4. Lỗi: Không tải được dữ liệu phòng hoặc báo lỗi mạng ở Frontend]
\textbf{Nguyên nhân:} Máy chủ Backend chưa được bật hoặc chưa chạy lệnh seeder dữ liệu.\\
\textbf{Cách xử lý:} Đảm bảo cửa sổ dòng lệnh Backend đang chạy lệnh \codeinline{php artisan serve} và đã chạy thành công lệnh \codeinline{php artisan migrate --seed}.
\end{warningbox}
