% ==================== BƯỚC 1: CHUẨN BỊ MÔI TRƯỜNG ====================
\section{BƯỚC 1: Chuẩn bị công cụ trên Windows (Yêu cầu duy nhất)}

Toàn bộ hệ thống đã được đóng gói hoàn chỉnh bằng công nghệ Container (\textbf{Docker}), do đó bạn \textbf{KHÔNG CẦN} cài đặt thủ công PHP, Composer, Node.js, MySQL hay MongoDB vào máy. Bạn chỉ cần cài đặt duy nhất phần mềm \textbf{Docker Desktop}.

\begin{stepbox}{1.1. Kiểm tra cấu hình máy tính Windows}
\begin{itemize}[leftmargin=5mm,itemsep=1pt]
  \item \textbf{Hệ điều hành:} Windows 10 (64-bit: Pro, Enterprise hoặc Home từ build 19041+) hoặc Windows 11.
  \item \textbf{Bộ nhớ RAM:} Tối thiểu 8 GB RAM (Khuyến nghị 16 GB để chạy mượt mà tất cả các dịch vụ).
  \item \textbf{Bật ảo hóa CPU (Virtualization):} Nhấn tổ hợp phím \codeinline{Ctrl + Shift + Esc} để mở \textbf{Task Manager} $\rightarrow$ chọn tab \textbf{Performance} $\rightarrow$ mục \textbf{CPU} $\rightarrow$ kiểm tra dòng \textbf{Virtualization: Enabled} (Nếu đang Disabled, hãy vào BIOS máy tính để bật Intel VT-x hoặc AMD-V).
\end{itemize}
\end{stepbox}

\begin{stepbox}{1.2. Tải và cài đặt Docker Desktop for Windows}
\begin{enumerate}[leftmargin=5mm,itemsep=1.5pt]
  \item Truy cập trang chủ chính thức của Docker: \url{https://www.docker.com/products/docker-desktop/}
  \item Bấm nút \textbf{"Download for Windows"} để tải file cài đặt \texttt{Docker Desktop Installer.exe}.
  \item Mở file vừa tải lên để tiến hành cài đặt:
    \begin{itemize}[itemsep=1pt]
      \item Tại màn hình tùy chọn, hãy \textbf{đánh dấu tích} vào ô: \textbf{"Use WSL 2 instead of Hyper-V (recommended)"}.
      \item Bấm \textbf{OK} và chờ tiến trình cài đặt hoàn tất.
    \end{itemize}
  \item Khi cài xong, nếu Docker yêu cầu khởi động lại máy tính (\textit{Restart}), hãy bấm \textbf{Restart Windows}.
  \item Sau khi máy bật lại, mở ứng dụng \textbf{Docker Desktop} từ màn hình Desktop hoặc menu Start.
  \item Chấp nhận điều khoản dịch vụ (\textit{Accept}) $\rightarrow$ Nhìn góc dưới bên trái của cửa sổ Docker Desktop, khi nào thấy \textbf{biểu tượng hình cá voi chuyển sang màu XANH LÁ CÂY (Engine running)} là Docker đã sẵn sàng hoạt động!
\end{enumerate}
\end{stepbox}

\begin{warningbox}[Xử lý khi Docker báo thiếu WSL 2 Linux Kernel]
Nếu Docker Desktop hiện thông báo \textit{"WSL 2 installation is incomplete"}, bạn chỉ cần mở \textbf{PowerShell (Run as Administrator)} và chạy lệnh sau:
\begin{center}
\codeinline{wsl --update} \quad hoặc \quad \codeinline{wsl --install}
\end{center}
Sau đó khởi động lại máy tính một lần nữa.
\end{warningbox}

% ==================== BƯỚC 2: TẢI MÃ NGUỒN ====================
\section{BƯỚC 2: Tải mã nguồn dự án về máy tính}

Bạn có thể chọn 1 trong 2 cách đơn giản sau để tải mã nguồn dự án:

\subsection*{Cách A: Tải file nén ZIP (Dễ nhất cho người mới bắt đầu)}
\begin{enumerate}[leftmargin=5mm,itemsep=1pt]
  \item Truy cập kho mã nguồn GitHub: \url{https://github.com/Phatjhhoq8/datphongkhachsan}
  \item Bấm nút màu xanh \textbf{"<> Code"} $\rightarrow$ Chọn \textbf{"Download ZIP"}.
  \item Giải nén file vừa tải vào một thư mục trên máy bạn (Ví dụ: \texttt{D:\textbackslash qlks} hoặc \texttt{C:\textbackslash datphongkhachsan}).
\end{enumerate}

\subsection*{Cách B: Sử dụng lệnh Git Clone}
Mở cửa sổ dòng lệnh (\textbf{Command Prompt} hoặc \textbf{PowerShell}) và chạy lệnh:

\begin{terminalcmd}{Sao chép mã nguồn từ GitHub về máy tính}
git clone https://github.com/Phatjhhoq8/datphongkhachsan.git
cd datphongkhachsan
\end{terminalcmd}

% ==================== BƯỚC 3: MỞ THƯ MỤC DỰ ÁN TRÊN WINDOWS ====================
\section{BƯỚC 3: Mở cửa sổ dòng lệnh tại thư mục dự án}

Để máy tính hiểu bạn muốn thao tác với dự án, hãy mở cửa sổ dòng lệnh ngay tại thư mục dự án:

\begin{tipbox}[Mẹo mở cửa sổ dòng lệnh siêu nhanh trong 2 giây trên Windows]
\begin{enumerate}[leftmargin=5mm,itemsep=1pt]
  \item Mở thư mục dự án bằng \textbf{File Explorer} (thư mục có chứa các file \texttt{docker-compose.yml}, \texttt{run.bat}, \texttt{backend}, \texttt{frontend}).
  \item Nhấp chuột trái vào \textbf{thanh địa chỉ đường dẫn} ở phía trên cùng của cửa sổ File Explorer (nơi hiện đường dẫn thư mục).
  \item Gõ chữ \codeinline{cmd} hoặc \codeinline{powershell} rồi nhấn phím \textbf{Enter}.
  \item Ngay lập tức, một cửa sổ dòng lệnh màu đen sẽ mở ra đúng ngay tại thư mục dự án của bạn!
\end{enumerate}
\end{tipbox}

% ==================== BƯỚC 4: KIỂM TRA FILE CẤU HÌNH .ENV ====================
\section{BƯỚC 4: Kiểm tra và chuẩn bị file cấu hình môi trường (.env)}

Thư mục dự án đã có sẵn file mẫu \texttt{.env.example}. Hãy chắc chắn bạn đã có file \texttt{.env}:

\begin{terminalcmd}{Tạo file .env từ file mẫu trên Windows (chạy trong CMD)}
copy .env.example .env
\end{terminalcmd}

\begin{terminalcmd}{Nếu bạn dùng PowerShell, chạy lệnh này để copy:}
Copy-Item .env.example .env
\end{terminalcmd}

\begin{notebox}[Các cổng kết nối mạng mặc định của hệ thống]
\begin{itemize}[leftmargin=5mm,itemsep=1pt]
  \item \textbf{Cổng 3000:} Dành cho giao diện Web người dùng (Frontend Vue 3).
  \item \textbf{Cổng 8000:} Dành cho máy chủ xử lý dữ liệu (Backend Laravel 12 API).
  \item \textbf{Cổng 3001:} Dành cho máy chủ thời gian thực (Realtime WebSocket Socket.IO).
  \item \textbf{Cổng 3306:} Dành cho Cơ sở dữ liệu MySQL 8.0.
  \item \textbf{Cổng 27017:} Dành cho Cơ sở dữ liệu MongoDB 8.0.
\end{itemize}
\textit{Ghi chú: Nếu máy tính bạn đã cài sẵn MySQL (như XAMPP/WampServer) đang chiếm cổng 3306, hãy mở file \texttt{.env} và đổi thành \codeinline{MYSQL\_HOST\_PORT=3307} hoặc tắt phần mềm XAMPP trước khi khởi chạy.}
\end{notebox}

% ==================== BƯỚC 5: KHỞI CHẠY DỰ ÁN ====================
\section{BƯỚC 5: Khởi chạy toàn bộ hệ thống (Chỉ 1 thao tác)}

Bạn có thể lựa chọn 1 trong 2 cách sau để khởi chạy:

\subsection*{Cách 1: Click đúp chuột vào file \texttt{run.bat} (Siêu đơn giản)}
\begin{enumerate}[leftmargin=5mm,itemsep=1pt]
  \item Tìm file có tên \textbf{\texttt{run.bat}} trong thư mục dự án.
  \item \textbf{Nhấp đúp chuột trái (Double-click)} vào file này.
  \item Một cửa sổ màu đen sẽ tự động hiện lên, kiểm tra Docker và thực hiện toàn bộ việc cài đặt khởi chạy!
\end{enumerate}

\subsection*{Cách 2: Chạy lệnh Docker Compose từ cửa sổ dòng lệnh (Chuẩn kỹ thuật)}
Tại cửa sổ CMD / PowerShell đã mở ở Bước 3, sao chép và dán lệnh sau:

\begin{terminalcmd}{Khởi chạy toàn bộ 10 container dịch vụ cùng một lúc}
docker compose up --build
\end{terminalcmd}

\begin{terminalcmd}{Hoặc nếu muốn chạy ngầm dưới nền (không chiếm cửa sổ dòng lệnh):}
docker compose up -d --build
\end{terminalcmd}

\begin{stepbox}{Hệ thống đang tự động làm những gì trong lần đầu chạy?}
Khi bạn chạy lệnh trên lần đầu tiên, hãy kiên nhẫn đợi từ \textbf{2 đến 5 phút} (tùy vào tốc độ mạng). Hệ thống sẽ tự động thực hiện hoàn toàn các công việc sau:
\begin{enumerate}[leftmargin=5mm,itemsep=1pt]
  \item Tự tải các hình ảnh Docker chính hãng: MySQL 8.0, MongoDB 8.0, Redis 7.0-alpine.
  \item Tự động xây dựng môi trường PHP 8.2+ cho Laravel Backend và Node.js 22 cho Realtime / Vue 3 Frontend.
  \item Khởi tạo Replica Set \texttt{rs0} cho MongoDB để hỗ trợ giao tác và stream.
  \item Tự động chạy lệnh tạo bảng CSDL (\textit{Migrations}) và nạp sẵn toàn bộ dữ liệu mẫu (\textit{Seeders}): Danh sách khách sạn, hình ảnh, phòng mẫu, dịch vụ, voucher và tài khoản demo.
  \item Khi bạn thấy dòng chữ thông báo các container đều ở trạng thái \textbf{Healthy} hoặc Frontend báo \texttt{Local: http://localhost:3000} là hệ thống đã hoàn toàn sẵn sàng!
\end{enumerate}
\end{stepbox}

% ==================== BƯỚC 7: CÁC CÂU LỆNH THƯỜNG DÙNG ====================
\section{BƯỚC 7: Các câu lệnh quản trị và vận hành thường dùng}

Mỗi khi cần thao tác bảo trì, hãy mở cửa sổ CMD / PowerShell tại thư mục dự án và dùng các lệnh sau:

\begin{terminalcmd}{Xem danh sách và trạng thái hoạt động của các dịch vụ}
docker compose ps
\end{terminalcmd}

\begin{terminalcmd}{Xem nhật ký (Logs) thời gian thực của hệ thống khi cần kiểm tra lỗi}
docker compose logs -f
\end{terminalcmd}

\begin{terminalcmd}{Tạm dừng và tắt toàn bộ hệ thống một cách an toàn}
docker compose down
\end{terminalcmd}

\begin{terminalcmd}{Bật lại hệ thống khi đã tải xong lần trước (khởi động cực nhanh)}
docker compose up -d
\end{terminalcmd}

\begin{terminalcmd}{Khôi phục / Nạp lại toàn bộ dữ liệu mẫu (Seeder) về trạng thái ban đầu}

\end{terminalcmd}

\begin{terminalcmd}{Chạy toàn bộ bộ kiểm thử tự động (Unit Test / Feature Test) của Backend}

\section{BƯỚC 8: Hướng dẫn xử lý các sự cố thường gặp trên Windows}

\begin{warningbox}[1. Lỗi: "docker: command not found" hoặc "error during connect"]
\textbf{Nguyên nhân:} Ứng dụng Docker Desktop chưa được bật hoặc đang trong quá trình khởi động.\\
\textbf{Cách xử lý:} Hãy mở ứng dụng \textbf{Docker Desktop} từ Start Menu, nhìn xuống góc dưới bên trái khi nào thấy biểu tượng cá voi chuyển sang \textbf{màu xanh lá cây (Running)} thì mới chạy lệnh.
\end{warningbox}

\begin{warningbox}[2. Lỗi: "Bind for 0.0.0.0:3306 failed: port is already allocated"]
\textbf{Nguyên nhân:} Cổng 3306 trên máy tính của bạn đã bị phần mềm khác chiếm dụng (thường do cài XAMPP, WampServer, hoặc dịch vụ MySQL cài trực tiếp trên Windows).\\
\textbf{Cách xử lý 1:} Tắt phần mềm XAMPP / MySQL Service trên Windows.\\
\textbf{Cách xử lý 2:} Mở file \texttt{.env} trong thư mục dự án, tìm dòng \texttt{MYSQL\_HOST\_PORT=3306} và đổi thành \codeinline{MYSQL\_HOST\_PORT=3307}, sau đó chạy lại lệnh \codeinline{docker compose up -d}.
\end{warningbox}

\begin{warningbox}[3. Lỗi: "Bind for 0.0.0.0:27017 failed: port is already allocated"]
\textbf{Nguyên nhân:} Cổng 27017 bị trùng do máy tính đã cài sẵn MongoDB Service.\\
\textbf{Cách xử lý:} Mở file \texttt{.env}, tìm dòng \texttt{MONGODB\_HOST\_PORT=27017} và đổi thành \codeinline{MONGODB\_HOST\_PORT=27018}, sau đó chạy lại lệnh.
\end{warningbox}

\begin{warningbox}[4. Lỗi: Trang web Frontend hiện màn hình trắng hoặc báo không kết nối được API]
\textbf{Nguyên nhân:} Trình duyệt chưa tải kịp hoặc Backend container đang trong quá trình nạp dữ liệu migration.\\
\textbf{Cách xử lý:} Đợi khoảng 30 giây rồi nhấn \codeinline{Ctrl + F5} trên trình duyệt để tải lại trang. Đảm bảo file \texttt{.env} có cấu hình \codeinline{VITE\_API\_BASE\_URL=http://localhost:8000/api/v1}.
\end{warningbox}
